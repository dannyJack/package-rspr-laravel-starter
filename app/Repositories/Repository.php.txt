    /**

     * @param array $attributes
     * @param array $exceptAttributes
     * @param array $orderAttributes
     * @param bool $withTrashed
     *
     * @return RepositoryResponseCollection
     */
    public function acquireAll(array $attributes = [], array $exceptAttributes = [], array $orderByAttributes = [], array $withRelationship = [], array $whereHas = [], bool $withTrashed = false): RepositoryResponseCollection
    {
        $rtn = $this->responseCollection();

        try {
            $rtn = $this->NTCacquireAll($attributes, $exceptAttributes, $orderByAttributes, $withRelationship, $whereHas, $withTrashed);
        } catch (\Exception $e) {
            \RSPRLog::error('Exception: ' . $e->getMessage());
        } catch (\Error $e) {
            \RSPRLog::error($e->getMessage());
        }

        return $rtn;
    }

    /**
     * acquire all model records
     * NTC (No Try Catch) method
     *
     * @param array $attributes
     * @param array $exceptAttributes
     * @param array $orderByAttributes
     * @param bool $withTrashed
     *
     * @return RepositoryResponseCollection
     */
    public function NTCacquireAll(array $attributes = [], array $exceptAttributes = [], array $orderByAttributes = [], array $withRelationship = [], array $whereHas = [], bool $withTrashed = false): RepositoryResponseCollection
    {
        $rtn = $this->responseCollection();

        if (!empty($this->model)) {
            $query = $this->model::query();
            $query = $this->conditionalQueries($query, $attributes, $exceptAttributes, $orderByAttributes, $withRelationship, $whereHas, $withTrashed);
            $rtn->data = $query->get();
            $rtn->success = true;
            $rtn->hasData = $rtn->data->count() > 0;
        }

        return $rtn;
    }

    {

    /**
     * acquire a model record
     * NTC (No Try Catch) method
     *
     * @param int $id
     * @param array $attributes
     * @param array $exceptAttributes
     * @param array $orderByAttributes
     * @param bool $withTrashed
     *
     * @return RepositoryResponseItem
     */
    public function NTCacquire(int $id, array $attributes = [], array $exceptAttributes = [], array $orderByAttributes = [], array $withRelationship = [], array $whereHas = [], bool $withTrashed = false): RepositoryResponseItem
    {
        $rtn = $this->responseItem();

        if (!empty($this->model)) {
            $query = $this->model::where(resolve($this->model)->getKeyName(), $id);
            $query = $this->conditionalQueries($query, $attributes, $exceptAttributes, $orderByAttributes, $withRelationship, $whereHas, $withTrashed);
            $data = $query->first();
            $rtn->success = true;

            if (empty($data)) {
                $rtn->data = $this->model::empty();
            } else {
                $rtn->hasData = true;
                $rtn->data = $data;
            }
        }

        return $rtn;
    }

    /**
     * add a model record
     * NTC (No Try Catch) method
     *
     * @param array $attributes
     * @return RepositoryResponseItem
     */
    public function NTCadd(array $attributes, array $withRelationship = []): RepositoryResponseItem
    {
        $rtn = $this->responseItem();

        if (!empty($this->model) && count($attributes) != 0) {
            $data = $this->model::create($attributes);

            if ($data) {
                $rtn->success = true;

                if (!empty($data)) {
                    $rtn->hasData = true;

                    if (is_null(resolve($this->model)->getPrimaryKey())) {
                        $rtn->data = $data->fresh();
                    } else {
                        $rtn->data = $this->NTCacquire($data->id, [], [], [], $withRelationship)->data;
                    }
                }
            }
        }

        \RSPRLog::responseCheck()->warning($rtn);

        return $rtn;
    }

    /**
     * add a list of model record
     * NTC (No Try Catch) method
     */
    public function NTCaddBulk(array $attributes): RepositoryResponseCollection
    {
        $rtn = $this->responseCollection();

        if (! empty($this->model) && count($attributes) != 0) {
            $lastId = null;
            $lastData = $this->model::orderBy(resolve($this->model)->getPrimaryKey(), 'desc')->limit(1)->get();

            if (count($lastData) > 0) {
                $lastId = $lastData[0]->id;
            }

            $data = $this->model::inserting($attributes);

            if ($data) {
                $rtn->success = true;
                $data = $this->model::query();

                if (! is_null($lastId)) {
                    $data = $data->where(resolve($this->model)->getPrimaryKey(), '>', $lastId);
                }

                $data = $data->limit(count($attributes))->get();

                if (! empty($data)) {
                    $rtn->hasData = true;
                    $rtn->data = Collection::make($data);
                }
            }
        }

        \RSPRLog::responseCheck()->warning($rtn);

        return $rtn;
    }

    public function adjust(int $id, array $attributes, array $withRelationship = []): RepositoryResponseItem
    {
        $rtn = $this->responseItem();

        try {
            $acquiredData = null;

            if (!empty($this->model)) {
                $acquireResponse = $this->NTCacquire($id);

                if ($acquireResponse->success && $acquireResponse->hasData) {
                    $rtn->hasData = true;
                    $rtn->data = $acquireResponse->data;
                    $acquiredData = $rtn->data;
                }
            }

            $rtn = $this->NTCadjust($id, $attributes, $withRelationship, $acquiredData);
        } catch (\Exception $e) {
            \RSPRLog::error('Exception: ' . $e->getMessage());
        } catch (\Error $e) {
            \RSPRLog::error($e->getMessage());
        }

        return $rtn;
    }

    /**
     * adjust a model record
     * NTC (No Try Catch) method
     *
     * @param int $id
     * @param array $attributes
     * @return RepositoryResponseItem
     */
    public function NTCadjust(int $id, array $attributes, array $withRelationship = [], Model $acquiredData = null): RepositoryResponseItem
    {
        $rtn = $this->responseItem();
        $hasAcquiredData = !empty($acquiredData);

        if (!empty($this->model) && count($attributes) != 0) {
            if (!$hasAcquiredData) {
                $acquireResponse = $this->NTCacquire($id, [], [], [], $withRelationship);

                if ($acquireResponse->success && $acquireResponse->hasData) {
                    $hasAcquiredData = true;
                    $acquiredData = $acquireResponse->data;
                }
            }

            if ($hasAcquiredData) {
                $rtn->hasData = true;
                $rtn->data = $acquiredData;
                $success = $acquiredData->update($attributes);

                if ($success) {
                    $rtn->success = true;
                    $rtn->data = $this->NTCacquire($acquireResponse->data->id, [], [], [], $withRelationship)->data;
                }
            }
        }

        \RSPRLog::responseCheck()->warning($rtn);

        return $rtn;
    }

     * NTC (No Try Catch) method
     */
    public function NTCannulBulkRecord(array $attributes, bool $hardDelete = null, bool $logResponseCheckWarning = true, $acquiredData = null): RepositoryResponseCollection
    {
        $rtn = $this->responseCollection();
        $hasAcquiredData = ! empty($acquiredData);

        if (! empty($this->model)) {
            if (! $hasAcquiredData) {
                $acquireAllResponse = $this->NTCacquireAll($attributes);

                if ($acquireAllResponse->success && $acquireAllResponse->hasData) {
                    $hasAcquiredData = true;
                    $acquiredData = $acquireAllResponse->data;
                }
            }

            if ($hasAcquiredData) {
                $rtn->hasData = true;
                $rtn->data = $acquiredData;
                $query = $this->model::query();
                $query = $this->conditionalQueries($query, $attributes);

                if (! $hardDelete && $this->model::isSoftDeletes()) {
                    $success = $query->update([
                        $this->model::DELETED_AT => $this->model::DELETE_FLAG_YES,
                    ]);
                } else {
                    $success = $query->delete();
                }

                if ($success) {
                    $rtn->success = true;
                }
            }
        }

        if ($logResponseCheckWarning) {
            \RSPRLog::responseCheck()->warning($rtn);
        }

        return $rtn;
    }

    /*======================================================================
    .* PRIVATE METHODS
    .*======================================================================*/
    
    protected function conditionalQueries(Builder $query, array $attributes = [], array $exceptAttributes = [], array $orderByAttributes = [], array $withRelationship = [], array $whereHas = [], bool $withTrashed = false): Builder
    {
        $query = $query->where(function ($query) use ($attributes) {
            foreach ($attributes as $key => $value) {
                if (is_array($value)) {
                    if (is_numeric($key)) {
                        if (is_numeric(array_key_first($value))) {
                            if (count($value) == 3) {
                                $query->where($value[0], $value[1], $value[2]);
                            } elseif (count($value) == 2) {
                                $query->where($value[0], $value[1]);
                            }
                        } else {
                            foreach ($value as $key2 => $value2) {
                                $query->where(function ($query) use ($key2, $value2) {
                                    $query->where($key2, $value2);

                                    return $query;
                                });
                            }
                        }
                    } else {
                        $query->whereIn($key, $value);
                    }
                } elseif (is_numeric($key) && is_string($value)) {
                    $query->whereRaw($value);
                } else {
                    $query->where($key, $value);
                }
            }

            return $query;
        });

        foreach ($exceptAttributes as $key => $value) {
            $query = $query->where($key, '!=', $value);
        }

        foreach ($orderByAttributes as $key => $value) {
            $query = $query->orderBy($key, $value);
        }

        foreach ($whereHas as $key => $value) {
            if (is_string($key) && is_callable($value)) {
                $query = $query->whereHas($key, $value);
            } else {
                $query = $query->whereHas($value);
            }
        }

        if ($withTrashed) {
            $query = $query->withTrashed();
        }

        if (count($withRelationship) > 0) {
            $query = $query->with($withRelationship);
        }

        return $query;
    }
